--- build.sh ---
#!/bin/bash

echo "[*] Setting up directories..."
mkdir -p build app app/backups

echo "[*] Cleaning old object files & Compiling HyperTex3D Ultimate..."
make clean
make

if [ $? -eq 0 ]; then
    echo "[+] Build Successful! Launching App..."
    ./app/hypertex3d
else
    echo "[-] Build Failed. Fix the errors above."
fi


--- Makefile ---
# File: Makefile
# Location: HyperTex3D/Makefile

CC = gcc
CFLAGS = -Wall -std=c99 -O3 -I./src
LDFLAGS = -lraylib -lGL -lm -lpthread -ldl -lrt -lX11

SRC = src/main.c src/engine.c src/ui.c src/fileio.c
OBJ = $(SRC:src/%.c=build/%.o)
OUT = app/hypertex3d

all: setup $(OUT)

setup:
	@mkdir -p build app app/backups

$(OUT): $(OBJ)
	$(CC) $(OBJ) -o $(OUT) $(LDFLAGS)

build/%.o: src/%.c src/hypertex.h
	$(CC) $(CFLAGS) -c $< -o $@

clean:
	rm -rf build/*.o app/hypertex3d

Color HexToC(const char* hex) {
    int r = 255, g = 255, b = 255, a = 255;
    if (strlen(hex) >= 7) sscanf(hex, "#%02x%02x%02x", &r, &g, &b);
    if (strlen(hex) == 9) sscanf(hex, "#%02x%02x%02x%02x", &r, &g, &b, &a);
    return (Color){ r, g, b, a };
}

static bool DrawStyledButton(Rectangle rec, const char* text, Color baseCol, Color textCol, int fontSize) {
    Vector2 m = GetMousePosition();
    bool hover = CheckCollisionPointRec(m, rec);
    Color drawCol = hover ? ColorBrightness(baseCol, 0.18f) : baseCol;
    DrawRectangleRec(rec, drawCol);
    DrawRectangleLinesEx(rec, 2, hover ? WHITE : ColorAlpha(BLACK, 0.5f));
    int tw = MeasureText(text, fontSize);
    DrawText(text, rec.x + (rec.width - tw) / 2, rec.y + (rec.height - fontSize) / 2, fontSize, textCol);
    return (hover && IsMouseButtonPressed(MOUSE_BUTTON_LEFT));
}

void DrawStartupUI(int w, int h, Camera3D* cam) {
    DrawRectangleGradientV(0, 0, w, h, (Color){ 14, 18, 28, 255 }, (Color){ 26, 33, 50, 255 });

    Rectangle card = { w/2 - 280, h/2 - 250, 560, 500 };
    DrawRectangleRec(card, (Color){ 22, 27, 40, 245 });
    DrawRectangleLinesEx(card, 2, (Color){ 65, 130, 255, 255 });

    DrawText("HYPERTEX 3D - PRO VOXEL STUDIO", w/2 - 220, card.y + 25, 24, WHITE);
    DrawText("Enter Workspace X, Y, Z Dimensions or Design a Character", w/2 - 210, card.y + 58, 14, LIGHTGRAY);

    const char* axisLabels[3] = { "WIDTH (X):", "HEIGHT (Y):", "DEPTH (Z):" };
    char* axisBuffers[3] = { inputX, inputY, inputZ };
    Color axisColors[3] = { (Color){230, 70, 70, 255}, (Color){60, 200, 100, 255}, (Color){70, 150, 255, 255} };

    for (int i = 0; i < 3; i++) {
        int bx = card.x + 50 + (i * 165);
        int by = card.y + 115;
        DrawText(axisLabels[i], bx, by - 22, 14, axisColors[i]);

        Rectangle box = { bx, by, 130, 44 };
        bool active = (activeStartupInput == i);
        DrawRectangleRec(box, active ? (Color){ 40, 50, 75, 255 } : (Color){ 15, 20, 30, 255 });
        DrawRectangleLinesEx(box, 2, active ? YELLOW : axisColors[i]);
        DrawText(axisBuffers[i], box.x + 16, box.y + 11, 22, WHITE);

        if (IsMouseButtonPressed(MOUSE_BUTTON_LEFT) && CheckCollisionPointRec(GetMousePosition(), box)) {
            activeStartupInput = i;
        }

        Rectangle btnMinus = { bx, by + 50, 62, 26 };
        Rectangle btnPlus  = { bx + 68, by + 50, 62, 26 };
        if (DrawStyledButton(btnMinus, "-4", (Color){45, 52, 70, 255}, WHITE, 14)) {
            int val = atoi(axisBuffers[i]);
            val = (val - 4 < 4) ? 4 : val - 4;
            sprintf(axisBuffers[i], "%d", val);
        }
        if (DrawStyledButton(btnPlus, "+4", (Color){45, 52, 70, 255}, WHITE, 14)) {
            int val = atoi(axisBuffers[i]);
            val = (val + 4 > 128) ? 128 : val + 4;
            sprintf(axisBuffers[i], "%d", val);
        }
    }

    int key = GetCharPressed();
    char* curBuf = (activeStartupInput == 0) ? inputX : (activeStartupInput == 1) ? inputY : inputZ;
    int len = strlen(curBuf);
    if (key >= '0' && key <= '9' && len < 3) {
        curBuf[len] = (char)key;
        curBuf[len + 1] = '\0';
    }
    if (IsKeyPressed(KEY_BACKSPACE) && len > 0) {
        curBuf[len - 1] = '\0';
    }
    if (IsKeyPressed(KEY_TAB)) {
        activeStartupInput = (activeStartupInput + 1) % 3;
    }

    Rectangle btnConfirm = { card.x + 50, card.y + 220, 460, 50 };
    if (DrawStyledButton(btnConfirm, "CONFIRM XYZ SIZE & START EDITOR (ENTER)", (Color){ 36, 160, 90, 255 }, WHITE, 17) || IsKeyPressed(KEY_ENTER)) {
        int sx = atoi(inputX), sy = atoi(inputY), sz = atoi(inputZ);
        if (sx <= 0) sx = 32;
        if (sy <= 0) sy = 32;
        if (sz <= 0) sz = 32;
        InitEngine(sx, sy, sz);
        cam->position = (Vector3){ gridX / 2.0f, gridY / 2.0f, fmax(gridX, gridZ) * 1.35f };
        cameraAngle.x = -PI / 2.0f;
        cameraAngle.y = 0.0f;
        appState = STATE_EDITOR;
        isMouseLocked = true;
        DisableCursor();
    }

    Rectangle btnChar = { card.x + 50, card.y + 288, 460, 56 };
    if (DrawStyledButton(btnChar, "DESIGN CHARACTER (16x16 Head | 11 Parts | 9 Joints)", (Color){ 215, 95, 25, 255 }, WHITE, 16)) {
        InitCharacterTemplate();
        charPanelPos = (Vector2){ (float)(w - 265), 56.0f };
        cam->position = (Vector3){ gridX / 2.0f, 38.0f, 68.0f };
        cameraAngle.x = -PI / 2.0f;
        cameraAngle.y = -0.05f;
        appState = STATE_EDITOR;
        isMouseLocked = true;
        DisableCursor();
    }
    DrawText("Includes breakable voxel skeleton: 4 Arm parts, 4 Leg parts, 2 Torso parts & 16x16 Head", card.x + 52, card.y + 350, 11, YELLOW);

    Rectangle btnBackup = { card.x + 50, card.y + 385, 460, 48 };
    char backupLabel[96];
    snprintf(backupLabel, sizeof(backupLabel), "OPEN AUTO-BACKUP (Recover Crash - %d / 30 Saved)", backupCount);
    if (DrawStyledButton(btnBackup, backupLabel, (Color){ 55, 90, 175, 255 }, WHITE, 16)) {
        LoadBackupIndex();
        prevStateBeforeBackup = STATE_STARTUP;
        appState = STATE_BACKUP_LIST;
    }

    DrawText("Auto-Backup runs every 5 seconds automatically in the background.", card.x + 95, card.y + 455, 12, GRAY);
}

void DrawBackupListUI(int w, int h, Camera3D* cam) {
    DrawRectangle(0, 0, w, h, (Color){ 14, 18, 28, 255 });
    DrawText("RECOVERY & AUTO-BACKUP MANAGER (LAST 30 BACKUPS - EVERY 5 SEC)", 40, 24, 22, WHITE);
    DrawText("Click any backup below to restore your exact 3D workspace immediately:", 40, 54, 15, LIGHTGRAY);

    Rectangle btnBack = { w - 180, 22, 140, 40 };
    if (DrawStyledButton(btnBack, "BACK (ESC)", MAROON, WHITE, 15) || IsKeyPressed(KEY_ESCAPE)) {
        appState = prevStateBeforeBackup;
        return;
    }

    if (backupCount == 0) {
        DrawText("No backups found yet. Backups are saved automatically every 5 seconds while editing.", 40, 140, 18, YELLOW);
        return;
    }

    for (int i = 0; i < backupCount; i++) {
        int col = i / 15;
        int row = i % 15;
        Rectangle rowRec = { 40 + col * 600, 90 + row * 39, 580, 34 };

        char itemText[256];
        snprintf(itemText, sizeof(itemText), "#%02d | %s | Size: %dx%dx%d | %s",
                 i + 1,
                 backupList[i].timestamp,
                 backupList[i].sx, backupList[i].sy, backupList[i].sz,
                 backupList[i].isCharMode ? "CHARACTER" : "CUSTOM XYZ");

        if (DrawStyledButton(rowRec, itemText, (i == 0) ? (Color){30, 95, 65, 255} : (Color){32, 40, 58, 255}, WHITE, 14)) {
            if (LoadProject(backupList[i].filename)) {
                cam->position = (Vector3){ gridX / 2.0f, gridY / 2.0f, fmax(gridX, gridZ) * 1.3f };
                cameraAngle.x = -PI / 2.0f;
                cameraAngle.y = 0.0f;
                appState = STATE_EDITOR;
                isMouseLocked = true;
                DisableCursor();
            }
        }
    }
}

void DrawCharacterPartPanel(int w, int h) {
    if (!isCharacterMode) return;

    float pw = 250.0f;
    float ph = 425.0f;
    Vector2 m = GetMousePosition();

    Rectangle headerBar = { charPanelPos.x, charPanelPos.y, pw, 30 };
    if (!isMouseLocked && !showFileMenu && IsMouseButtonPressed(MOUSE_BUTTON_LEFT) && CheckCollisionPointRec(m, headerBar)) {
        isDraggingCharPanel = true;
        charPanelDragOffset = (Vector2){ m.x - charPanelPos.x, m.y - charPanelPos.y };
    }
    if (isDraggingCharPanel) {
        if (IsMouseButtonDown(MOUSE_BUTTON_LEFT)) {
            charPanelPos.x = m.x - charPanelDragOffset.x;
            charPanelPos.y = m.y - charPanelDragOffset.y;
            if (charPanelPos.x < 4) charPanelPos.x = 4;
            if (charPanelPos.y < 46) charPanelPos.y = 46;
            if (charPanelPos.x + pw > w - 4) charPanelPos.x = w - pw - 4;
            if (charPanelPos.y + ph > h - 4) charPanelPos.y = h - ph - 4;
        } else {
            isDraggingCharPanel = false;
        }
    }

    Rectangle panel = { charPanelPos.x, charPanelPos.y, pw, ph };
    DrawRectangleRec(panel, (Color){ 16, 20, 32, 240 });
    DrawRectangleRec(headerBar, isDraggingCharPanel ? (Color){ 235, 115, 30, 255 } : (Color){ 185, 80, 20, 255 });
    DrawRectangleLinesEx(panel, 2, (Color){ 235, 125, 45, 255 });

    DrawText("CHARACTER PARTS [DRAG]", panel.x + 10, panel.y + 8, 13, WHITE);
    DrawText("Click Part to Solo | Eye = Full Hide:", panel.x + 10, panel.y + 35, 11, LIGHTGRAY);

    Rectangle btnAll = { panel.x + 8, panel.y + 52, pw - 16, 24 };
    if (DrawStyledButton(btnAll, "SHOW ALL PARTS (RESET)", (activePartFilter == 0) ? (Color){35, 135, 80, 255} : (Color){36, 44, 64, 255}, WHITE, 12)) {
        activePartFilter = 0;
        for (int k = 1; k <= NUM_BODY_PARTS; k++) partHidden[k] = false;
        UpdateCulling();
    }

    for (int i = 1; i <= NUM_BODY_PARTS; i++) {
        float ry = panel.y + 54 + i * 26;
        Rectangle soloBtn = { panel.x + 8, ry, pw - 74, 23 };
        Rectangle hideBtn = { panel.x + pw - 62, ry, 54, 23 };

        bool isSolo = (activePartFilter == i);
        bool isHid  = partHidden[i];

        Color partBg = isHid ? (Color){ 25, 28, 38, 255 } : (isSolo ? (Color){ 210, 95, 20, 255 } : (Color){ 34, 42, 62, 255 });
        if (DrawStyledButton(soloBtn, bodyPartNames[i], partBg, isHid ? GRAY : WHITE, 11)) {
            if (activePartFilter == i) {
                activePartFilter = 0;
            } else {
                activePartFilter = i;
                partHidden[i] = false;
            }
            UpdateCulling();
        }

        if (DrawStyledButton(hideBtn, isHid ? "HIDDEN" : "SHOW", isHid ? MAROON : (Color){35, 125, 75, 255}, WHITE, 10)) {
            partHidden[i] = !partHidden[i];
            if (partHidden[i] && activePartFilter == i) activePartFilter = 0;
            UpdateCulling();
        }
    }

    Rectangle jointBtn = { panel.x + 8, panel.y + 370, pw - 16, 28 };
    if (DrawStyledButton(jointBtn, showJointsOverlay ? "9 JOINTS RIG: VISIBLE" : "9 JOINTS RIG: HIDDEN",
                         showJointsOverlay ? (Color){40, 140, 90, 255} : DARKGRAY, WHITE, 12)) {
        showJointsOverlay = !showJointsOverlay;
    }
    DrawText("Hidden parts are 100% invisible", panel.x + 28, panel.y + 404, 11, YELLOW);
}

void HandleTopHUDClicks(void) {
    Vector2 m = GetMousePosition();

    if (CheckCollisionPointRec(m, (Rectangle){ 10, 6, 95, 32 })) {
        showFileMenu = !showFileMenu;
        return;
    }

    if (showFileMenu) {
        if (m.x >= 14 && m.x <= 236 && m.y >= 44 && m.y <= 282) {
            int idx = (int)(m.y - 44) / 34;
            if (idx == 0) { SaveProject("app/project.ht3d"); }
            else if (idx == 1) { showSaveAsPopup = true; isTypingPath = true; isMouseLocked = false; EnableCursor(); }
            else if (idx == 2) { showExportPopup = true; isTypingPath = true; isMouseLocked = false; EnableCursor(); }
            else if (idx == 3) { appState = STATE_IMPORT_PATH; isTypingPath = true; isMouseLocked = false; EnableCursor(); }
            else if (idx == 4) { LoadProject("app/project.ht3d"); }
            else if (idx == 5) { LoadBackupIndex(); prevStateBeforeBackup = STATE_EDITOR; appState = STATE_BACKUP_LIST; isMouseLocked = false; EnableCursor(); }
            else if (idx == 6) { appState = STATE_STARTUP; isMouseLocked = false; EnableCursor(); }
        }
        showFileMenu = false;
    }
}

void HandlePopupInputs(void) {
    if (IsKeyPressed(KEY_ESCAPE)) {
        showExportPopup = false;
        showSaveAsPopup = false;
        if (appState == STATE_IMPORT_PATH) appState = STATE_EDITOR;
        isMouseLocked = true;
        DisableCursor();
        return;
    }

    char* targetStr = showExportPopup ? exportFileName : (showSaveAsPopup ? saveAsFileName : importFileName);

    if (isTypingPath) {
        int key = GetCharPressed();
        int len = strlen(targetStr);
        if ((key >= 32) && (key <= 125) && (len < 100)) {
            targetStr[len] = (char)key;
            targetStr[len + 1] = '\0';
        }
        if (IsKeyPressed(KEY_BACKSPACE) && len > 0) {
            targetStr[len - 1] = '\0';
        }

        if (IsKeyPressed(KEY_ENTER)) {
            if (showExportPopup && strlen(exportFileName) > 0) {
                char path[160];
                snprintf(path, sizeof(path), "app/%s", exportFileName);
                ExportToPNG(path);
                showExportPopup = false;
                isMouseLocked = true;
                DisableCursor();
            } else if (showSaveAsPopup && strlen(saveAsFileName) > 0) {
                if (!strstr(saveAsFileName, ".ht3d")) strcat(saveAsFileName, ".ht3d");
                char path[160];
                snprintf(path, sizeof(path), "app/%s", saveAsFileName);
                SaveProject(path);
                showSaveAsPopup = false;
                isMouseLocked = true;
                DisableCursor();
            } else if (appState == STATE_IMPORT_PATH && strlen(importFileName) > 0) {
                if (importImage.data != NULL) UnloadImage(importImage);
                if (importTexture.id != 0) UnloadTexture(importTexture);
                importImage = LoadImage(importFileName);
                if (importImage.data != NULL) {
                    importTexture = LoadTextureFromImage(importImage);
                    importSelectRec = (Rectangle){0, 0, importImage.width, importImage.height};
                    sprintf(cropX, "0"); sprintf(cropY, "0");
                    sprintf(cropW, "%d", importImage.width); sprintf(cropH, "%d", importImage.height);
                    appState = STATE_IMPORT_2D;
                } else {
                    SetStatus("Failed to load PNG file!");
                    appState = STATE_EDITOR;
                    isMouseLocked = true;
                    DisableCursor();
                }
            }
        }
    }
}

void HandleImport2DInput(void) {

            str[len - 1] = '\0';
        }
        importSelectRec.x = atoi(cropX);
        importSelectRec.y = atoi(cropY);
        importSelectRec.width = atoi(cropW);
        importSelectRec.height = atoi(cropH);
    }

    if (importTexture.id != 0 && !activeCropInput) {
        int w = GetScreenWidth(), h = GetScreenHeight();
        float scale = fmin((float)(w - 300) / importTexture.width, (float)h / importTexture.height) * 0.8f;
        Vector2 offset = { (w - 300)/2 - (importTexture.width * scale)/2, h/2 - (importTexture.height * scale)/2 };
        Vector2 mousePos = GetMousePosition();

        if (IsMouseButtonPressed(MOUSE_BUTTON_LEFT) && mousePos.x < w - 300) {
            importDragStart = mousePos;
            isImportDragging = true;
        }
        if (isImportDragging) {
            float rx = (fmin(importDragStart.x, mousePos.x) - offset.x) / scale;
            float ry = (fmin(importDragStart.y, mousePos.y) - offset.y) / scale;
            float rw = fabs(importDragStart.x - mousePos.x) / scale;
            float rh = fabs(importDragStart.y - mousePos.y) / scale;

            if (rx < 0) rx = 0;
            if (ry < 0) ry = 0;
            if (rx + rw > importImage.width) rw = importImage.width - rx;
            if (ry + rh > importImage.height) rh = importImage.height - ry;

            importSelectRec = (Rectangle){rx, ry, rw, rh};
            sprintf(cropX, "%d", (int)rx); sprintf(cropY, "%d", (int)ry);
            sprintf(cropW, "%d", (int)rw); sprintf(cropH, "%d", (int)rh);
        }
        if (IsMouseButtonReleased(MOUSE_BUTTON_LEFT)) isImportDragging = false;
    }
}

RayCollision GetCustomRayCollisionPlane(Ray ray, Vector3 planePos, Vector3 planeNormal) {
    RayCollision result = {0};
    float denom = Vector3DotProduct(ray.direction, planeNormal);
    if (fabs(denom) > 0.000001f) {
        float t = Vector3DotProduct(Vector3Subtract(planePos, ray.position), planeNormal) / denom;
        if (t >= 0) {
            result.hit = true;
            result.distance = t;
            result.point = Vector3Add(ray.position, Vector3Scale(ray.direction, t));
            result.normal = planeNormal;
        }
    }
    return result;
}

int main(void) {
    const int w = 1280, h = 720;
    SetConfigFlags(FLAG_MSAA_4X_HINT | FLAG_WINDOW_RESIZABLE);
    InitWindow(w, h, "HyperTex3D Ultimate - Voxel & Character Studio");
    SetExitKey(KEY_NULL);
    SetTargetFPS(60);

    LoadBackupIndex();
    lastBackupTime = GetTime();

    Camera3D camera = { 0 };
    camera.up = (Vector3){ 0.0f, 1.0f, 0.0f };
    camera.fovy = 60.0f;
    camera.projection = CAMERA_PERSPECTIVE;

    bool isDragging = false;
    Vector3 dragStart = {0, 0, 0};

    while (!WindowShouldClose()) {
        int sw = GetScreenWidth();
        int sh = GetScreenHeight();

        if (statusTimer > 0.0f) {
            statusTimer -= GetFrameTime();
        }

        if (appState == STATE_EDITOR && grid != NULL && (GetTime() - lastBackupTime >= 5.0)) {
            PerformAutoBackup();
            lastBackupTime = GetTime();
        }

        if (IsFileDropped()) {
            FilePathList droppedFiles = LoadDroppedFiles();
            if (droppedFiles.count > 0 && IsFileExtension(droppedFiles.paths[0], ".png")) {
                if (importImage.data != NULL) UnloadImage(importImage);
                if (importTexture.id != 0) UnloadTexture(importTexture);
                importImage = LoadImage(droppedFiles.paths[0]);
                if (importImage.data != NULL) {
                    importTexture = LoadTextureFromImage(importImage);
                    importSelectRec = (Rectangle){0, 0, importImage.width, importImage.height};
                    sprintf(cropX, "0"); sprintf(cropY, "0");
                    sprintf(cropW, "%d", importImage.width); sprintf(cropH, "%d", importImage.height);
                    appState = STATE_IMPORT_2D;
                    isMouseLocked = false;
                    EnableCursor();
                }
            }
            UnloadDroppedFiles(droppedFiles);
        }

        if (appState == STATE_STARTUP) {
            BeginDrawing();
            DrawStartupUI(sw, sh, &camera);
            EndDrawing();
            continue;
        }

        if (appState == STATE_BACKUP_LIST) {
            BeginDrawing();
            DrawBackupListUI(sw, sh, &camera);
            EndDrawing();
            continue;
        }

        if (appState == STATE_IMPORT_PATH) {
            HandlePopupInputs();
            BeginDrawing();
            DrawImportPathPopup(sw, sh);
            EndDrawing();
            continue;
        }

        if (appState == STATE_IMPORT_2D) {
            HandleImport2DInput();
            if (IsKeyPressed(KEY_ESCAPE)) {
                appState = STATE_EDITOR;
                isMouseLocked = true;
                DisableCursor();
            } else if (IsKeyPressed(KEY_ENTER)) {
                CreateClipboardFromImage();
                appState = STATE_EDITOR;
                isPasting = true;
                pasteMode = 0;
                pastePos = (Vector3){ gridX / 2, gridY / 2, gridZ / 2 };
                isMouseLocked = false;
                EnableCursor();
            }
            BeginDrawing();
            DrawImport2DScreen(sw, sh);
            EndDrawing();
            continue;
        }

        if (showSelectPopup)  { BeginDrawing(); DrawSelectPopup(sw, sh);       EndDrawing(); continue; }
        if (showToolsPopup)   { BeginDrawing(); DrawToolsPopup(sw, sh);        EndDrawing(); continue; }
        if (showSlot4Palette) { BeginDrawing(); DrawSlot4PalettePopup(sw, sh); EndDrawing(); continue; }
        if (showExportPopup)  { HandlePopupInputs(); BeginDrawing(); DrawExportPopup(sw, sh); EndDrawing(); continue; }
        if (showSaveAsPopup)  { HandlePopupInputs(); BeginDrawing(); DrawSaveAsPopup(sw, sh); EndDrawing(); continue; }

        if (IsKeyPressed(KEY_Q) && !showInventory) {
            isMouseLocked = !isMouseLocked;
            if (isMouseLocked) DisableCursor(); else EnableCursor();
        }

        if (IsKeyPressed(KEY_E)) {
            showInventory = !showInventory;
            if (showInventory) {
                isMouseLocked = false;
                EnableCursor();
                hexBoxActive = false;
            } else {
                isMouseLocked = true;
                DisableCursor();
                if (currentTool == TOOL_PICKER) currentTool = TOOL_BRUSH;
            }
        }
        if (showInventory) HandleInventoryInput();

        if (IsKeyPressed(KEY_ONE)) { showToolsPopup = true; isMouseLocked = false; EnableCursor(); }
        if (IsKeyPressed(KEY_TWO)) {
            hpState = (hpState + 1) % 4;
            if (hpState == 1) hpPos = fmax(0, fmin((int)camera.position.y, gridY - 1));
            if (hpState == 2) hpPos = fmax(0, fmin((int)camera.position.x, gridX - 1));
            if (hpState == 3) hpPos = fmax(0, fmin((int)camera.position.z, gridZ - 1));
        }
        if (IsKeyPressed(KEY_THREE)) { selColorIdx = 2; currentTool = TOOL_BRUSH; }

        if (IsKeyPressed(KEY_FOUR)) {
            selColorIdx = 3;
            currentTool = TOOL_BRUSH;
            showSlot4Palette = true;
            isMouseLocked = false;
            EnableCursor();
        }

        for (int i = 4; i < 9; i++) {
            if (IsKeyPressed(KEY_ONE + i)) { selColorIdx = i; currentTool = TOOL_BRUSH; }
        }
        if (IsKeyPressed(KEY_ZERO)) { selColorIdx = 9; currentTool = TOOL_BRUSH; }

        if (IsKeyDown(KEY_LEFT_CONTROL) && IsKeyPressed(KEY_S)) {
            SaveProject("app/project.ht3d");
        }
        if (IsKeyDown(KEY_LEFT_CONTROL) && IsKeyPressed(KEY_V) && clipboard != NULL) {
            isPasting = true;
            pasteMode = 0;
            pastePos = (Vector3){ gridX / 2, gridY / 2, gridZ / 2 };
            isMouseLocked = false;
            EnableCursor();
        }

        Vector3 lookForward = {0};
        if (!isMouseLocked && !showInventory && !showToolsPopup && IsMouseButtonDown(MOUSE_BUTTON_RIGHT)) {
            Vector2 md = GetMouseDelta();
            cameraAngle.x += md.x * 0.003f;
            cameraAngle.y -= md.y * 0.003f;
            if (cameraAngle.y > 1.5f) cameraAngle.y = 1.5f;
            if (cameraAngle.y < -1.5f) cameraAngle.y = -1.5f;
        }

        if ((isMouseLocked || IsMouseButtonDown(MOUSE_BUTTON_RIGHT)) && !showInventory && !showToolsPopup) {
            if (isMouseLocked) {
                Vector2 md = GetMouseDelta();
                cameraAngle.x += md.x * 0.003f;
                cameraAngle.y -= md.y * 0.003f;
                if (cameraAngle.y > 1.5f) cameraAngle.y = 1.5f;
                if (cameraAngle.y < -1.5f) cameraAngle.y = -1.5f;
            }
            lookForward.x = cos(cameraAngle.y) * cos(cameraAngle.x);
            lookForward.y = sin(cameraAngle.y);
            lookForward.z = cos(cameraAngle.y) * sin(cameraAngle.x);

            Vector3 moveForward = Vector3Normalize((Vector3){ cos(cameraAngle.x), 0.0f, sin(cameraAngle.x) });
            Vector3 moveRight = { -moveForward.z, 0.0f, moveForward.x };

            float dt = GetFrameTime() * 14.0f * moveSpeed;
            if (IsKeyDown(KEY_W)) camera.position = Vector3Add(camera.position, Vector3Scale(moveForward, dt));
            if (IsKeyDown(KEY_S)) camera.position = Vector3Subtract(camera.position, Vector3Scale(moveForward, dt));
            if (IsKeyDown(KEY_D)) camera.position = Vector3Add(camera.position, Vector3Scale(moveRight, dt));
            if (IsKeyDown(KEY_A)) camera.position = Vector3Subtract(camera.position, Vector3Scale(moveRight, dt));
            if (IsKeyDown(KEY_SPACE)) camera.position.y += dt;
            if (IsKeyDown(KEY_LEFT_SHIFT)) camera.position.y -= dt;
        } else {
            lookForward.x = cos(cameraAngle.y) * cos(cameraAngle.x);
            lookForward.y = sin(cameraAngle.y);
            lookForward.z = cos(cameraAngle.y) * sin(cameraAngle.x);
        }
        camera.target = Vector3Add(camera.position, lookForward);

        Ray ray = isMouseLocked ? (Ray){camera.position, lookForward} : GetMouseRay(GetMousePosition(), camera);
        float closestDist = 10000.0f;
        Vector3 hitNorm = {0};
        int hitX = -1, hitY = -1, hitZ = -1, placeX = -1, placeY = -1, placeZ = -1;
        bool blockHit = false;

        if (!isMouseLocked && !showInventory && !showToolsPopup) {
            if (isPasting) {
                Vector3 cPos = { pastePos.x + clipSize.x/2.0f, pastePos.y + clipSize.y/2.0f, pastePos.z + clipSize.z/2.0f };
                BoundingBox boxX = { (Vector3){cPos.x, cPos.y-0.5f, cPos.z-0.5f}, (Vector3){cPos.x+4, cPos.y+0.5f, cPos.z+0.5f} };
                BoundingBox boxY = { (Vector3){cPos.x-0.5f, cPos.y, cPos.z-0.5f}, (Vector3){cPos.x+0.5f, cPos.y+4, cPos.z+0.5f} };
                BoundingBox boxZ = { (Vector3){cPos.x-0.5f, cPos.y-0.5f, cPos.z}, (Vector3){cPos.x+0.5f, cPos.y+0.5f, cPos.z+4} };

                int hover = 0;
                if (GetRayCollisionBox(ray, boxX).hit) hover = 1;
                else if (GetRayCollisionBox(ray, boxY).hit) hover = 2;
                else if (GetRayCollisionBox(ray, boxZ).hit) hover = 3;

                if (IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) activeGizmo = hover;
                if (IsMouseButtonReleased(MOUSE_BUTTON_LEFT)) activeGizmo = 0;

                if (activeGizmo != 0) {
                    Vector3 pNorm = {0};
                    if (activeGizmo == 1) pNorm = (fabs(camera.position.y - cPos.y) > fabs(camera.position.z - cPos.z)) ? (Vector3){0,1,0} : (Vector3){0,0,1};
                    if (activeGizmo == 2) pNorm = (fabs(camera.position.x - cPos.x) > fabs(camera.position.z - cPos.z)) ? (Vector3){1,0,0} : (Vector3){0,0,1};
                    if (activeGizmo == 3) pNorm = (fabs(camera.position.x - cPos.x) > fabs(camera.position.y - cPos.y)) ? (Vector3){1,0,0} : (Vector3){0,1,0};

                    RayCollision coll = GetCustomRayCollisionPlane(ray, cPos, pNorm);
                    if (coll.hit) {
                        if (activeGizmo == 1) pastePos.x = round(coll.point.x - clipSize.x/2.0f);
                        if (activeGizmo == 2) pastePos.y = round(coll.point.y - clipSize.y/2.0f);
                        if (activeGizmo == 3) pastePos.z = round(coll.point.z - clipSize.z/2.0f);
                    }
                }
                if (IsKeyPressed(KEY_ENTER)) {
                    if (pasteMode == 0) pasteMode = 1;
                    else { CommitPaste(); isPasting = false; isMouseLocked = true; DisableCursor(); }
                }
                if (pasteMode == 1 && (IsKeyPressed(KEY_LEFT) || IsKeyPressed(KEY_RIGHT))) RotateClipboardY();
            }
        }

        // Voxel Raycast
        for (int z = 0; z < gridZ; z++) {
            for (int y = 0; y < gridY; y++) {
                for (int x = 0; x < gridX; x++) {
                    Voxel v = grid[GetIndex(x, y, z)];
                    if (v.active && v.exposedMask != 0 && IsPartVisible(v.partId)) {
                        BoundingBox box = { (Vector3){x, y, z}, (Vector3){x + 1, y + 1, z + 1} };
                        RayCollision col = GetRayCollisionBox(ray, box);
                        if (col.hit && col.distance > 0.01f && col.distance < closestDist) {
                            closestDist = col.distance;
                            hitX = x; hitY = y; hitZ = z;
                            hitNorm = col.normal;
                            blockHit = true;
                        }
                    }
                }
            }
        }

        if (!blockHit) {
            float tMin = 10000.0f;
            Vector3 cp = {0};
            if (hpState != 0) {
                if (hpState == 2 && ray.direction.x != 0) {
                    float t = (hpPos - ray.position.x) / ray.direction.x;
                    if (t > 0) { cp = Vector3Add(ray.position, Vector3Scale(ray.direction, t)); if (cp.y>=0 && cp.y<gridY && cp.z>=0 && cp.z<gridZ) { placeX=hpPos; placeY=(int)cp.y; placeZ=(int)cp.z; } }
                } else if (hpState == 1 && ray.direction.y != 0) {
                    float t = (hpPos - ray.position.y) / ray.direction.y;
                    if (t > 0) { cp = Vector3Add(ray.position, Vector3Scale(ray.direction, t)); if (cp.x>=0 && cp.x<gridX && cp.z>=0 && cp.z<gridZ) { placeX=(int)cp.x; placeY=hpPos; placeZ=(int)cp.z; } }
                } else if (hpState == 3 && ray.direction.z != 0) {
                    float t = (hpPos - ray.position.z) / ray.direction.z;
                    if (t > 0) { cp = Vector3Add(ray.position, Vector3Scale(ray.direction, t)); if (cp.x>=0 && cp.x<gridX && cp.y>=0 && cp.y<gridY) { placeX=(int)cp.x; placeY=(int)cp.y; placeZ=hpPos; } }
                }
            } else {
                if (ray.direction.y != 0) {
                    float t = (0 - ray.position.y) / ray.direction.y;
                    if (t > 0 && t < tMin) {
                        cp = Vector3Add(ray.position, Vector3Scale(ray.direction, t));
                        if (cp.x >= 0 && cp.x < gridX && cp.z >= 0 && cp.z < gridZ) {
                            tMin = t; placeX = (int)cp.x; placeY = 0; placeZ = (int)cp.z;
                        }
                    }
                }
            }
        } else {
            placeX = hitX + (int)hitNorm.x;
            placeY = hitY + (int)hitNorm.y;
            placeZ = hitZ + (int)hitNorm.z;
        }

        if (!isPasting && isMouseLocked) {
            if (IsMouseButtonPressed(MOUSE_BUTTON_LEFT) && blockHit) {
                grid[GetIndex(hitX, hitY, hitZ)].active = false;
                UpdateCulling();
            }

            if (currentTool == TOOL_PICKER && IsMouseButtonPressed(MOUSE_BUTTON_RIGHT) && blockHit) {
                hotbar[selColorIdx] = grid[GetIndex(hitX, hitY, hitZ)].color;
                currentTool = TOOL_BRUSH;
            } else if (currentTool == TOOL_BRUSH) {
                if (IsMouseButtonPressed(MOUSE_BUTTON_RIGHT) && InBounds(placeX, placeY, placeZ)) {
                    int idx = GetIndex(placeX, placeY, placeZ);
                    grid[idx].active = true;
                    grid[idx].color = hotbar[selColorIdx];
                    if (isCharacterMode) {
                        grid[idx].partId = (activePartFilter > 0) ? activePartFilter : (blockHit ? grid[GetIndex(hitX, hitY, hitZ)].partId : 1);
                    }
                    UpdateCulling();
                }
            } else {
                if (IsMouseButtonPressed(MOUSE_BUTTON_RIGHT) && (blockHit || InBounds(placeX, placeY, placeZ))) {
                    isDragging = true;
                    dragStart = blockHit ? (Vector3){hitX, hitY, hitZ} : (Vector3){placeX, placeY, placeZ};
                }
                if (IsMouseButtonReleased(MOUSE_BUTTON_RIGHT) && isDragging) {
                    isDragging = false;
                    if (currentTool == TOOL_SELECT) {
                        selStart = dragStart;
                        selEnd = blockHit ? (Vector3){hitX, hitY, hitZ} : (Vector3){placeX, placeY, placeZ};
                        showSelectPopup = true;
                        isMouseLocked = false;
                        EnableCursor();
                    } else {
                        ApplyTool((int)dragStart.x, (int)dragStart.y, (int)dragStart.z,
                                  blockHit ? hitX : placeX, blockHit ? hitY : placeY, blockHit ? hitZ : placeZ,
                                  currentTool, hotbar[selColorIdx]);
                    }
                }
            }
        }

        if (!isMouseLocked && !showInventory && !showSelectPopup && !showToolsPopup && !showSlot4Palette && !showExportPopup && !showSaveAsPopup && IsMouseButtonPressed(MOUSE_BUTTON_LEFT)) {
            HandleTopHUDClicks();
        }

        // ==================== RENDER 3D SCENE ====================
        BeginDrawing();
        ClearBackground((Color){ 16, 20, 30, 255 });
        BeginMode3D(camera);

        DrawCubeWires((Vector3){gridX / 2.0f, gridY / 2.0f, gridZ / 2.0f}, gridX, gridY, gridZ, ColorAlpha(SKYBLUE, 0.22f));
        for (int x = 0; x <= gridX; x += 2) {
            DrawLine3D((Vector3){x, 0, 0}, (Vector3){x, 0, gridZ}, ColorAlpha(GRAY, 0.2f));
        }
        for (int z = 0; z <= gridZ; z += 2) {
            DrawLine3D((Vector3){0, 0, z}, (Vector3){gridX, 0, z}, ColorAlpha(GRAY, 0.2f));
        }

        for (int z = 0; z < gridZ; z++) {
            for (int y = 0; y < gridY; y++) {
                for (int x = 0; x < gridX; x++) {
                    Voxel v = grid[GetIndex(x, y, z)];
                    if (v.active && v.exposedMask != 0 && IsPartVisible(v.partId)) {
                        DrawVoxelShaded(x, y, z, v);
                    }
                }
            }
        }

        if (isPasting && clipboard) {
            for (int i = 0; i < clipCount; i++) {
                Vector3 gPos = { pastePos.x + clipboard[i].dx + 0.5f, pastePos.y + clipboard[i].dy + 0.5f, pastePos.z + clipboard[i].dz + 0.5f };
                DrawCube(gPos, 1.0f, 1.0f, 1.0f, ColorAlpha(clipboard[i].color, 0.65f));
            }
        }

        if (!isPasting) {
            if (isDragging) {
                int cX = blockHit ? hitX : placeX;
                int cY = blockHit ? hitY : placeY;
                int cZ = blockHit ? hitZ : placeZ;
                Vector3 center = { (dragStart.x + cX)/2.0f + 0.5f, (dragStart.y + cY)/2.0f + 0.5f, (dragStart.z + cZ)/2.0f + 0.5f };
                float wBox = fabs(dragStart.x - cX) + 1.0f;
                float hBox = fabs(dragStart.y - cY) + 1.0f;
                float lBox = fabs(dragStart.z - cZ) + 1.0f;
                DrawCubeWires(center, wBox, hBox, lBox, (currentTool == TOOL_SELECT) ? LIME : ORANGE);
            } else {
                if (blockHit) {
                    DrawCubeWires((Vector3){hitX + 0.5f, hitY + 0.5f, hitZ + 0.5f}, 1.03f, 1.03f, 1.03f, YELLOW);
                }
                if (InBounds(placeX, placeY, placeZ) && currentTool != TOOL_SELECT && currentTool != TOOL_PICKER && !blockHit) {
                    DrawCubeWires((Vector3){placeX + 0.5f, placeY + 0.5f, placeZ + 0.5f}, 1.0f, 1.0f, 1.0f, ColorAlpha(hotbar[selColorIdx], 0.75f));
                }
            }
        }

        rlDisableDepthTest();
        if (isCharacterMode && showJointsOverlay) {
            for (int j = 0; j < NUM_JOINTS; j++) {
                DrawSphere(charJoints[j].pos, 0.65f, GOLD);
                DrawSphereWires(charJoints[j].pos, 0.75f, 6, 6, ORANGE);
            }
        }

        if (hpState != 0) {
            if (hpState == 2) {
                for (int i = 0; i <= gridY; i += 2) DrawLine3D((Vector3){hpPos, i, 0}, (Vector3){hpPos, i, gridZ}, ColorAlpha(RED, 0.5f));
            } else if (hpState == 1) {
                for (int i = 0; i <= gridX; i += 2) DrawLine3D((Vector3){i, hpPos, 0}, (Vector3){i, hpPos, gridZ}, ColorAlpha(GREEN, 0.5f));
            } else if (hpState == 3) {
                for (int i = 0; i <= gridX; i += 2) DrawLine3D((Vector3){i, 0, hpPos}, (Vector3){i, gridY, hpPos}, ColorAlpha(BLUE, 0.5f));
            }
        }
        rlEnableDepthTest();
        EndMode3D();

        if (isCharacterMode && showJointsOverlay && !showInventory) {
            for (int j = 0; j < NUM_JOINTS; j++) {
                if (Vector3DotProduct(lookForward, Vector3Subtract(charJoints[j].pos, camera.position)) > 0) {
                    Vector2 sp = GetWorldToScreen(charJoints[j].pos, camera);
                    DrawText(charJoints[j].name, (int)sp.x + 8, (int)sp.y - 6, 11, GOLD);
                }
            }
        }

        // Top HUD rendered last so menus are never occluded
        if (!showInventory && appState == STATE_EDITOR) {
            DrawCharacterPartPanel(sw, sh);
            DrawBottomHotbar(sw, sh);
            DrawTopHUD(sw, sh, blockHit ? hitX : placeX, blockHit ? hitY : placeY, blockHit ? hitZ : placeZ);

            if (isMouseLocked) {
                DrawRectangle(sw/2 - 1, sh/2 - 8, 2, 16, WHITE);
                DrawRectangle(sw/2 - 8, sh/2 - 1, 16, 2, WHITE);
            }
        }

        if (showInventory) DrawInventory(sw, sh);
        DrawFPS(sw - 95, sh - 28);
        EndDrawing();
    }

    if (grid != NULL) free(grid);
    if (clipboard != NULL) free(clipboard);
    CloseWindow();
    return 0;
}

// Helper to paint a 3D box unwrapped to standard 2D layout
// Box dimensions: dx, dy, dz
// Layout inside canvas at (dstX, dstY):
//   Row 1: [Top (dx*dz)] [Bottom (dx*dz)]
//   Row 2: [Right (dz*dy)] [Front (dx*dy)] [Left (dz*dy)] [Back (dx*dy)]
static void UnwrapBoxPartToImage(Image* img, int minX, int minY, int minZ, int dx, int dy, int dz, int dstX, int dstY) {
    // 1. Top face (+Y): x in 0..dx-1, z in 0..dz-1
    for (int x = 0; x < dx; x++) {
        for (int z = 0; z < dz; z++) {
            int gx = minX + x, gy = minY + dy - 1, gz = minZ + z;
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + dz + x, dstY + z, v.color);
                }
            }
        }
    }

    // 2. Bottom face (-Y)
    for (int x = 0; x < dx; x++) {
        for (int z = 0; z < dz; z++) {
            int gx = minX + x, gy = minY, gz = minZ + z;
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + dz + dx + x, dstY + z, v.color);
                }
            }
        }
    }

    // 3. Right face (+X): z in 0..dz-1, y in 0..dy-1
    for (int z = 0; z < dz; z++) {
        for (int y = 0; y < dy; y++) {
            int gx = minX + dx - 1, gy = minY + (dy - 1 - y), gz = minZ + (dz - 1 - z);
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + z, dstY + dz + y, v.color);
                }
            }
        }
    }

    // 4. Front face (+Z): x in 0..dx-1, y in 0..dy-1
    for (int x = 0; x < dx; x++) {
        for (int y = 0; y < dy; y++) {
            int gx = minX + x, gy = minY + (dy - 1 - y), gz = minZ + dz - 1;
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + dz + x, dstY + dz + y, v.color);
                }
            }
        }
    }

    // 5. Left face (-X)
    for (int z = 0; z < dz; z++) {
        for (int y = 0; y < dy; y++) {
            int gx = minX, gy = minY + (dy - 1 - y), gz = minZ + z;
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + dz + dx + z, dstY + dz + y, v.color);
                }
            }
        }
    }

    // 6. Back face (-Z)
    for (int x = 0; x < dx; x++) {
        for (int y = 0; y < dy; y++) {
            int gx = minX + (dx - 1 - x), gy = minY + (dy - 1 - y), gz = minZ;
            if (InBounds(gx, gy, gz)) {
                Voxel v = grid[GetIndex(gx, gy, gz)];
                if (v.active && v.color.a > 0) {
                    ImageDrawPixel(img, dstX + dz + dx + dz + x, dstY + dz + y, v.color);
                }
            }
        }
    }
}

void ExportToPNG(const char* filename) {
    if (!grid) return;

    // Check if character mode skin export is selected
    if (isCharacterMode && exportAsCharacterSkin) {
        char cleanPath[256];
        strncpy(cleanPath, filename, 255);
        char* ext = strstr(cleanPath, ".png");
        if (ext) *ext = '\0';
        ExportCharacterSkinSheet(cleanPath);
        return;
    }

    // Default Voxel Slices Export
    int imgWidth = gridX * gridZ;
    int imgHeight = gridY;

    Image exportImg = GenImageColor(imgWidth, imgHeight, BLANK);

    for (int z = 0; z < gridZ; z++) {
        for (int y = 0; y < gridY; y++) {
            for (int x = 0; x < gridX; x++) {
                int idx = GetIndex(x, y, z);
                if (grid[idx].active && grid[idx].color.a > 0) {
                    if (!IsPartVisible(grid[idx].partId)) continue;
                    int imgX = (z * gridX) + x;
                    int imgY = (gridY - 1) - y;
                    ImageDrawPixel(&exportImg, imgX, imgY, grid[idx].color);
                }
            }
        }
    }

    ExportImage(exportImg, filename);
    UnloadImage(exportImg);

    char msg[128];
    snprintf(msg, sizeof(msg), "Exported PNG (%dx%d): %s", imgWidth, imgHeight, filename);
    SetStatus(msg);
}

void SaveProject(const char* filename) {
    if (!grid) return;
    FILE* file = fopen(filename, "wb");
    if (file) {
        unsigned int magic = HT3D_MAGIC;
        int charFlag = isCharacterMode ? 1 : 0;
        fwrite(&magic, sizeof(unsigned int), 1, file);
        fwrite(&gridX, sizeof(int), 1, file);
        fwrite(&gridY, sizeof(int), 1, file);
        fwrite(&gridZ, sizeof(int), 1, file);
        fwrite(&charFlag, sizeof(int), 1, file);
        fwrite(grid, sizeof(Voxel), (size_t)gridX * gridY * gridZ, file);
        fclose(file);

        char msg[128];
        snprintf(msg, sizeof(msg), "Saved Project: %s", filename);
        SetStatus(msg);
    }
}

bool LoadProject(const char* filename) {
    FILE* file = fopen(filename, "rb");
    if (!file) {
        SetStatus("Error: Could not open file!");
        return false;
    }

    unsigned int magic = 0;
    fread(&magic, sizeof(unsigned int), 1, file);

    if (magic == HT3D_MAGIC) {
        int nx = 16, ny = 16, nz = 16, charFlag = 0;
        fread(&nx, sizeof(int), 1, file);
        fread(&ny, sizeof(int), 1, file);
        fread(&nz, sizeof(int), 1, file);
        fread(&charFlag, sizeof(int), 1, file);

        if (charFlag) {
            InitCharacterTemplate();
        }
        InitEngine(nx, ny, nz);
        isCharacterMode = (charFlag != 0);
        fread(grid, sizeof(Voxel), (size_t)gridX * gridY * gridZ, file);
    } else {
        int legacySize = (int)magic;
        if (legacySize >= 4 && legacySize <= 128) {
            InitEngine(legacySize, legacySize, legacySize);
            fread(grid, sizeof(Voxel), (size_t)legacySize * legacySize * legacySize, file);
        }
    }

    fclose(file);
    UpdateCulling();

    char msg[128];
    snprintf(msg, sizeof(msg), "Loaded: %s (%dx%dx%d)", filename, gridX, gridY, gridZ);
    SetStatus(msg);
    return true;
}

static void SaveBackupIndex(void) {
    FILE* f = fopen("app/backups/index.dat", "wb");
    if (f) {
        fwrite(&backupCount, sizeof(int), 1, f);
        fwrite(backupList, sizeof(BackupEntry), backupCount, f);
        fclose(f);
    }
}

void LoadBackupIndex(void) {
    mkdir("app", 0777);
    mkdir("app/backups", 0777);
    FILE* f = fopen("app/backups/index.dat", "rb");
    if (f) {
        fread(&backupCount, sizeof(int), 1, f);
        if (backupCount < 0) backupCount = 0;
        if (backupCount > MAX_BACKUPS) backupCount = MAX_BACKUPS;
        fread(backupList, sizeof(BackupEntry), backupCount, f);
        fclose(f);
    } else {
        backupCount = 0;
    }
}

void PerformAutoBackup(void) {
    if (!grid) return;
    mkdir("app", 0777);
    mkdir("app/backups", 0777);

    if (backupCount >= MAX_BACKUPS) {
        remove(backupList[MAX_BACKUPS - 1].filename);
        backupCount = MAX_BACKUPS - 1;
    }

    for (int i = backupCount; i > 0; i--) {
        backupList[i] = backupList[i - 1];
    }

    time_t now = time(NULL);
    struct tm* t = localtime(&now);

    BackupEntry entry;
    strftime(entry.timestamp, sizeof(entry.timestamp), "%Y-%m-%d  %H:%M:%S", t);
    snprintf(entry.filename, sizeof(entry.filename), "app/backups/backup_%ld_%d.ht3d", (long)now, rand() % 1000);
    entry.sx = gridX;
    entry.sy = gridY;
    entry.sz = gridZ;
    entry.isCharMode = isCharacterMode;

    FILE* file = fopen(entry.filename, "wb");
    if (file) {
        unsigned int magic = HT3D_MAGIC;
        int charFlag = isCharacterMode ? 1 : 0;
        fwrite(&magic, sizeof(unsigned int), 1, file);
        fwrite(&gridX, sizeof(int), 1, file);
        fwrite(&gridY, sizeof(int), 1, file);
        fwrite(&gridZ, sizeof(int), 1, file);
        fwrite(&charFlag, sizeof(int), 1, file);
        fwrite(grid, sizeof(Voxel), (size_t)gridX * gridY * gridZ, file);
        fclose(file);

        backupList[0] = entry;
        backupCount++;
        SaveBackupIndex();
    }
}

int GetIndex(int x, int y, int z) {
    return (z * gridX * gridY) + (y * gridX) + x;
}

bool InBounds(int x, int y, int z) {
    return (x >= 0 && x < gridX && y >= 0 && y < gridY && z >= 0 && z < gridZ);
}

bool IsPartVisible(unsigned char partId) {
    if (!isCharacterMode) return true;
    if (partId >= 1 && partId <= NUM_BODY_PARTS) {
        if (partHidden[partId]) return false;
    }
    if (activePartFilter != 0 && partId != activePartFilter) {
        return false;
    }
    return true;
}

void InitEngine(int sx, int sy, int sz) {
    if (sx < 4) sx = 4; if (sx > 128) sx = 128;
    if (sy < 4) sy = 4; if (sy > 128) sy = 128;
    if (sz < 4) sz = 4; if (sz > 128) sz = 128;

    gridX = sx; gridY = sy; gridZ = sz;
    sprintf(inputX, "%d", gridX);
    sprintf(inputY, "%d", gridY);
    sprintf(inputZ, "%d", gridZ);

    if (grid != NULL) {
        free(grid);
        grid = NULL;
    }

    size_t total = (size_t)gridX * gridY * gridZ;
    grid = (Voxel*)malloc(total * sizeof(Voxel));
    for (size_t i = 0; i < total; i++) {
        grid[i].active = false;
        grid[i].exposedMask = 0;
        grid[i].partId = 0;
        grid[i].color = BLANK;
    }
    isCharacterMode = false;
    activePartFilter = 0;
    for (int i = 0; i <= NUM_BODY_PARTS; i++) partHidden[i] = false;
}

static void FillBoxPart(int x1, int y1, int z1, int x2, int y2, int z2, unsigned char partId, Color baseCol) {
    for (int x = x1; x <= x2; x++) {
        for (int y = y1; y <= y2; y++) {
            for (int z = z1; z <= z2; z++) {
                if (InBounds(x, y, z)) {
                    int idx = GetIndex(x, y, z);
                    grid[idx].active = true;
                    grid[idx].partId = partId;
                    int checker = ((x + y + z) % 2 == 0) ? 6 : -6;
                    Color c = baseCol;
                    c.r = (unsigned char)fmin(255, fmax(0, (int)c.r + checker));
                    c.g = (unsigned char)fmin(255, fmax(0, (int)c.g + checker));
                    c.b = (unsigned char)fmin(255, fmax(0, (int)c.b + checker));
                    grid[idx].color = c;
                }
            }
        }
    }
}

static bool IsOccluder(int x, int y, int z, unsigned char selfPart) {

    if (!v.active || v.color.a < 255) return false;
    if (!IsPartVisible(v.partId)) return false;
    if (isCharacterMode && v.partId != selfPart) return false;
    return true;
}

void UpdateCulling(void) {
    if (!grid) return;
    for (int z = 0; z < gridZ; z++) {
        for (int y = 0; y < gridY; y++) {
            for (int x = 0; x < gridX; x++) {
                int idx = GetIndex(x, y, z);
                if (!grid[idx].active || !IsPartVisible(grid[idx].partId)) {
                    grid[idx].exposedMask = 0;
                    continue;
                }
                unsigned char pid = grid[idx].partId;
                unsigned char mask = 0;
                if (!IsOccluder(x + 1, y, z, pid)) mask |= 1;  // +X
                if (!IsOccluder(x - 1, y, z, pid)) mask |= 2;  // -X
                if (!IsOccluder(x, y + 1, z, pid)) mask |= 4;  // +Y
                if (!IsOccluder(x, y - 1, z, pid)) mask |= 8;  // -Y
                if (!IsOccluder(x, y, z + 1, pid)) mask |= 16; // +Z
                if (!IsOccluder(x, y, z - 1, pid)) mask |= 32; // -Z
                grid[idx].exposedMask = mask;
            }
        }
    }
}

static Color ShadeColor(Color c, float factor) {
    if (!smoothShading) return c;
    return (Color){
        (unsigned char)fmin(255.0f, c.r * factor),
        (unsigned char)fmin(255.0f, c.g * factor),
        (unsigned char)fmin(255.0f, c.b * factor),
        c.a
    };
}

void DrawVoxelShaded(int x, int y, int z, Voxel v) {
    if (v.exposedMask == 0 || !IsPartVisible(v.partId)) return;

    Color base = v.color;
    if (base.a == 0) {
        DrawCubeWires((Vector3){ x + 0.5f, y + 0.5f, z + 0.5f }, 1.0f, 1.0f, 1.0f, Fade(RED, 0.4f));
        return;
    }

    float fx = (float)x, fy = (float)y, fz = (float)z;
    Color cTop    = ShadeColor(base, 1.05f);
    Color cBottom = ShadeColor(base, 0.52f);
    Color cFront  = ShadeColor(base, 0.92f);
    Color cBack   = ShadeColor(base, 0.78f);
    Color cRight  = ShadeColor(base, 0.84f);
    Color cLeft   = ShadeColor(base, 0.68f);

    rlBegin(RL_QUADS);
    if (v.exposedMask & 4) {
        rlColor4ub(cTop.r, cTop.g, cTop.b, cTop.a);
        rlVertex3f(fx,     fy + 1, fz);
        rlVertex3f(fx,     fy + 1, fz + 1);
        rlVertex3f(fx + 1, fy + 1, fz + 1);
        rlVertex3f(fx + 1, fy + 1, fz);
    }
    if (v.exposedMask & 8) {
        rlColor4ub(cBottom.r, cBottom.g, cBottom.b, cBottom.a);
        rlVertex3f(fx,     fy, fz);
        rlVertex3f(fx + 1, fy, fz);
        rlVertex3f(fx + 1, fy, fz + 1);
        rlVertex3f(fx,     fy, fz + 1);
    }
    if (v.exposedMask & 16) {
        rlColor4ub(cFront.r, cFront.g, cFront.b, cFront.a);
        rlVertex3f(fx,     fy,     fz + 1);
        rlVertex3f(fx + 1, fy,     fz + 1);
        rlVertex3f(fx + 1, fy + 1, fz + 1);
        rlVertex3f(fx,     fy + 1, fz + 1);
    }
    if (v.exposedMask & 32) {
        rlColor4ub(cBack.r, cBack.g, cBack.b, cBack.a);
        rlVertex3f(fx + 1, fy,     fz);
        rlVertex3f(fx,     fy,     fz);
        rlVertex3f(fx,     fy + 1, fz);
        rlVertex3f(fx + 1, fy + 1, fz);
    }
    if (v.exposedMask & 1) {
        rlColor4ub(cRight.r, cRight.g, cRight.b, cRight.a);
        rlVertex3f(fx + 1, fy,     fz + 1);
        rlVertex3f(fx + 1, fy,     fz);
        rlVertex3f(fx + 1, fy + 1, fz);
        rlVertex3f(fx + 1, fy + 1, fz + 1);
    }
    if (v.exposedMask & 2) {
        rlColor4ub(cLeft.r, cLeft.g, cLeft.b, cLeft.a);
        rlVertex3f(fx, fy,     fz);
        rlVertex3f(fx, fy,     fz + 1);
        rlVertex3f(fx, fy + 1, fz + 1);
        rlVertex3f(fx, fy + 1, fz);
    }
    rlEnd();

    if (showWireframe) {
        DrawCubeWires((Vector3){ fx + 0.5f, fy + 0.5f, fz + 0.5f }, 1.0f, 1.0f, 1.0f, ColorAlpha(BLACK, 0.22f));
    }
}

void CopySelection(bool deleteOriginal) {
    if (clipboard != NULL) { free(clipboard); clipboard = NULL; }
    int minX = fmin(selStart.x, selEnd.x), maxX = fmax(selStart.x, selEnd.x);
    int minY = fmin(selStart.y, selEnd.y), maxY = fmax(selStart.y, selEnd.y);
    int minZ = fmin(selStart.z, selEnd.z), maxZ = fmax(selStart.z, selEnd.z);

    clipSize = (Vector3){ maxX - minX + 1, maxY - minY + 1, maxZ - minZ + 1 };
    clipCount = 0;

    for (int x = minX; x <= maxX; x++) {
        for (int y = minY; y <= maxY; y++) {
            for (int z = minZ; z <= maxZ; z++) {
                if (InBounds(x, y, z)) {
                    Voxel v = grid[GetIndex(x, y, z)];
                    if (v.active && IsPartVisible(v.partId)) clipCount++;
                }
            }
        }
    }

    if (clipCount == 0) return;

    clipboard = (ClipVoxel*)malloc(clipCount * sizeof(ClipVoxel));
    int i = 0;
    for (int x = minX; x <= maxX; x++) {
        for (int y = minY; y <= maxY; y++) {
            for (int z = minZ; z <= maxZ; z++) {
                if (InBounds(x, y, z)) {
                    int idx = GetIndex(x, y, z);
                    if (grid[idx].active && IsPartVisible(grid[idx].partId)) {
                        clipboard[i].dx = x - minX;
                        clipboard[i].dy = y - minY;
                        clipboard[i].dz = z - minZ;
                        clipboard[i].partId = grid[idx].partId;
                        clipboard[i].color = grid[idx].color;
                        if (deleteOriginal) grid[idx].active = false;
                        i++;
                    }
                }
            }
        }
    }
    if (deleteOriginal) UpdateCulling();
}

void CreateClipboardFromImage(void) {
    if (clipboard != NULL) { free(clipboard); clipboard = NULL; }
    clipCount = 0;
    int sX = (int)importSelectRec.x, sY = (int)importSelectRec.y;
    int rw = (int)importSelectRec.width, rh = (int)importSelectRec.height;

    if (sX < 0) sX = 0;
    if (sY < 0) sY = 0;
    if (sX + rw > importImage.width) rw = importImage.width - sX;
    if (sY + rh > importImage.height) rh = importImage.height - sY;

    for (int y = 0; y < rh; y++) {
        for (int x = 0; x < rw; x++) {
            Color c = GetImageColor(importImage, sX + x, sY + y);
            if (c.a > 0) clipCount++;
        }
    }
    if (clipCount == 0) return;

    clipboard = (ClipVoxel*)malloc(clipCount * sizeof(ClipVoxel));
    clipSize = (Vector3){ rw, rh, 1 };

    int i = 0;
    for (int y = 0; y < rh; y++) {
        for (int x = 0; x < rw; x++) {
            Color c = GetImageColor(importImage, sX + x, sY + y);
            if (c.a > 0) {
                clipboard[i].dx = x;
                clipboard[i].dy = (rh - 1 - y);
                clipboard[i].dz = 0;
                clipboard[i].partId = (activePartFilter > 0) ? activePartFilter : 1;
                clipboard[i].color = c;
                i++;
            }
        }
    }
}

void DeleteSelection(void) {
    int minX = fmin(selStart.x, selEnd.x), maxX = fmax(selStart.x, selEnd.x);
    int minY = fmin(selStart.y, selEnd.y), maxY = fmax(selStart.y, selEnd.y);
    int minZ = fmin(selStart.z, selEnd.z), maxZ = fmax(selStart.z, selEnd.z);
    for (int x = minX; x <= maxX; x++) {
        for (int y = minY; y <= maxY; y++) {
            for (int z = minZ; z <= maxZ; z++) {
                if (InBounds(x, y, z)) {
                    int idx = GetIndex(x, y, z);
                    if (IsPartVisible(grid[idx].partId)) grid[idx].active = false;
                }
            }
        }
    }
    UpdateCulling();
}

void RotateClipboardY(void) {
    if (!clipboard || clipCount == 0) return;
    float newWidth = clipSize.z;
    float newDepth = clipSize.x;
    for (int i = 0; i < clipCount; i++) {
        int oldX = clipboard[i].dx;
        int oldZ = clipboard[i].dz;
        clipboard[i].dx = (int)(newWidth - 1) - oldZ;
        clipboard[i].dz = oldX;
    }
    clipSize.x = newWidth;
    clipSize.z = newDepth;
}

void CommitPaste(void) {
    if (!clipboard) return;
    for (int i = 0; i < clipCount; i++) {
        int wX = (int)pastePos.x + clipboard[i].dx;
        int wY = (int)pastePos.y + clipboard[i].dy;
        int wZ = (int)pastePos.z + clipboard[i].dz;
        if (InBounds(wX, wY, wZ)) {
            int idx = GetIndex(wX, wY, wZ);
            grid[idx].active = true;
            grid[idx].partId = (activePartFilter > 0) ? activePartFilter : clipboard[i].partId;
            grid[idx].color = clipboard[i].color;
        }
    }
    UpdateCulling();
}

void ApplyTool(int sX, int sY, int sZ, int eX, int eY, int eZ, Tool tool, Color c) {
    unsigned char targetPart = (activePartFilter > 0) ? activePartFilter : (isCharacterMode ? 1 : 0);

    if (tool == TOOL_LINE) {
        float dx = eX - sX, dy = eY - sY, dz = eZ - sZ;
        float steps = fmax(fabs(dx), fmax(fabs(dy), fabs(dz)));
        for (int i = 0; i <= steps; i++) {
            float t = (steps == 0) ? 0.0f : (float)i / steps;
            int x = round(sX + t * dx), y = round(sY + t * dy), z = round(sZ + t * dz);
            if (InBounds(x, y, z)) {
                int idx = GetIndex(x, y, z);
                grid[idx].active = true;
                grid[idx].partId = targetPart;
                grid[idx].color = c;
            }
        }
    } else {
        int minX = fmin(sX, eX), maxX = fmax(sX, eX);
        int minY = fmin(sY, eY), maxY = fmax(sY, eY);
        int minZ = fmin(sZ, eZ), maxZ = fmax(sZ, eZ);

        for (int x = minX; x <= maxX; x++) {
            for (int y = minY; y <= maxY; y++) {
                for (int z = minZ; z <= maxZ; z++) {
                    if (!InBounds(x, y, z)) continue;
                    int idx = GetIndex(x, y, z);
                    if (tool == TOOL_RECT_2D) {
                        if (y != sY) continue;
                        if (x == minX || x == maxX || z == minZ || z == maxZ) {
                            grid[idx].active = true;
                            grid[idx].partId = targetPart;
                            grid[idx].color = c;
                        }
                    } else if (tool == TOOL_BRUSH) {
                        grid[idx].active = true;
                        grid[idx].partId = targetPart;
                        grid[idx].color = c;
                    }
                }
            }
        }
    }
    UpdateCulling();
}
